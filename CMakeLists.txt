cmake_minimum_required(VERSION 3.25)
project(dsp56300_core VERSION 0.1.0 LANGUAGES C CXX)

include(cmake/CompilerCache.cmake)
include(CTest)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
foreach(language C CXX)
  if(NOT CMAKE_${language}_COMPILER_ID STREQUAL "Clang" OR
     NOT CMAKE_${language}_COMPILER_VERSION VERSION_GREATER_EQUAL 19 OR
     NOT CMAKE_${language}_COMPILER_VERSION VERSION_LESS 20)
    message(FATAL_ERROR "Use LLVM 19: cmake --preset strict")
  endif()
endforeach()
include(cmake/Quality.cmake)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
get_filename_component(DSP56300_LLVM_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
find_program(DSP56300_CLANG_TIDY NAMES clang-tidy-19 clang-tidy
  HINTS "${DSP56300_LLVM_BIN}" REQUIRED)
find_program(DSP56300_CLANG_FORMAT NAMES clang-format-19 clang-format
  HINTS "${DSP56300_LLVM_BIN}" REQUIRED)
foreach(tool IN ITEMS DSP56300_CLANG_TIDY DSP56300_CLANG_FORMAT)
  execute_process(COMMAND "${${tool}}" --version OUTPUT_VARIABLE tool_version
    COMMAND_ERROR_IS_FATAL ANY)
  if(NOT tool_version MATCHES "version 19\\.")
    message(FATAL_ERROR "${tool} must be LLVM 19")
  endif()
endforeach()
option(DSP56300_BUILD_TESTING "Build DSP library tests" ${PROJECT_IS_TOP_LEVEL})

set(CMAKE_CXX_CLANG_TIDY "${DSP56300_CLANG_TIDY};--warnings-as-errors=*")

set(dsp56300_semantic_sources
  src/agu.cpp src/arithmetic.cpp src/bit_fields.cpp src/bits.cpp
  src/cache_control.cpp src/conditional.cpp src/control.cpp
  src/core.cpp src/debugging.cpp src/decode.cpp src/exceptions.cpp
  src/external_memory.cpp src/fetch.cpp src/immediate.cpp src/instruction_shape.cpp
  src/integer.cpp src/iteration.cpp src/loop.cpp
  src/move.cpp src/processing.cpp src/products.cpp
  src/program_cache.cpp src/registers.cpp src/repeat.cpp src/runtime.cpp
  src/saturation.cpp src/semantic_region.cpp src/semantic.cpp src/image_runtime.cpp src/shifts.cpp src/state.cpp
  src/jit/llvm/region_memory.cpp
  src/transfer.cpp src/transfer_description.cpp src/transfer_info.cpp
  src/transfer_parallel.cpp src/transfer_special.cpp src/viterbi.cpp)
add_library(dsp56300_semantics OBJECT ${dsp56300_semantic_sources})
add_library(dsp56300_core STATIC
  src/io.cpp src/image_compile.cpp src/image_artifact.cpp src/image_entry.cpp src/jit/preparation.cpp src/jit/image_plan.cpp
  src/jit/prefixes.cpp src/jit/prefix_access.cpp src/jit/llvm/runtime.cpp
  src/jit/llvm/lifecycle.cpp src/jit/llvm/bindings.cpp
  src/jit/llvm/image.cpp src/jit/llvm/image_regions.cpp src/jit/llvm/image_specialize.cpp
  src/jit/llvm/region.cpp src/jit/llvm/region_plan.cpp src/jit/llvm/alu.cpp
  src/jit/move.cpp src/jit/registers.cpp)
add_library(dsp56300::core ALIAS dsp56300_core)
target_include_directories(dsp56300_core PUBLIC include)
target_compile_features(dsp56300_core PUBLIC cxx_std_23)
target_link_libraries(dsp56300_core PUBLIC dsp56300_quality PRIVATE dsp56300_semantics)
include(cmake/Llvm.cmake)
set(dsp56300_own_targets dsp56300_core dsp56300_semantics)
set(DSP56300_SEMANTIC_TARGET dsp56300_semantics)
# Retain public Io call boundaries until consumer binding, without changing native helpers.
add_library(dsp56300_io_ir OBJECT src/io.cpp)
target_include_directories(dsp56300_io_ir PRIVATE include src)
target_link_libraries(dsp56300_io_ir PRIVATE dsp56300_quality)
target_compile_definitions(dsp56300_io_ir PRIVATE DSP56300_BINDING_IR)
target_compile_options(dsp56300_io_ir PRIVATE -emit-llvm)
set_target_properties(dsp56300_io_ir PROPERTIES INTERPROCEDURAL_OPTIMIZATION_RELEASE OFF)
list(APPEND dsp56300_own_targets dsp56300_io_ir)
if(DSP56300_SANITIZER STREQUAL "address-undefined")
  target_compile_options(dsp56300_io_ir PRIVATE -Xclang -disable-llvm-passes)
  # Keep frontend UBSan checks; defer LLVM optimization/ASan until the final
  # generated functions exist. Native helpers retain their normal instrumentation.
  add_library(dsp56300_semantic_ir_objects OBJECT ${dsp56300_semantic_sources})
  list(APPEND dsp56300_own_targets dsp56300_semantic_ir_objects)
  set(DSP56300_SEMANTIC_TARGET dsp56300_semantic_ir_objects)
  target_compile_options(dsp56300_semantic_ir_objects PRIVATE
    -O3 -flto=thin -Xclang -disable-llvm-passes)
endif()
foreach(own_target IN LISTS dsp56300_own_targets)
  target_include_directories(${own_target} PRIVATE include src src/jit src/jit/x64)
  target_include_directories(${own_target} SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
  target_link_libraries(${own_target} PRIVATE dsp56300_quality)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
    target_compile_options(${own_target} PRIVATE $<$<CONFIG:Release>:-mcpu=native>)
  else()
    target_compile_options(${own_target} PRIVATE $<$<CONFIG:Release>:-march=native>)
  endif()
endforeach()
# Normal helpers use production O3/ThinLTO with all requested diagnostics.
target_compile_options(dsp56300_semantics PRIVATE -O3 -flto=thin)
include(cmake/SemanticIr.cmake)
include(cmake/EmbedSemantic.cmake)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
  target_compile_options(dsp56300_core INTERFACE $<$<CONFIG:Release>:-mcpu=native>)
else()
  target_compile_options(dsp56300_core INTERFACE $<$<CONFIG:Release>:-march=native>)
endif()
if(NOT APPLE)
  target_link_options(dsp56300_core PUBLIC -fuse-ld=lld)
endif()

  "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.hpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/test/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/test/*.hpp")

  COMMAND "${DSP56300_CLANG_FORMAT}" --dry-run --Werror ${DSP56300_OWN_FILES}
  VERBATIM)
